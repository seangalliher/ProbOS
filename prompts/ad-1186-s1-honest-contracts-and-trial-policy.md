## 1. Problem

- `src/probos/__main__.py:1849-1851`: `probos qa run-contracts` evaluates through `_stub_invoker`, which returns `""`. A must-rule case fails on that reply, reading an unwired runtime as the agent's failure. A must_not-only case passes on it, reading as a pass a reply no agent gave. `tests/test_better_agents_behavior_contract.py:164-182` pins the false pass (`assert rc == 0`, "trivially passes").
- `src/probos/cognitive/behavior_contract.py:133-139` records an invoker exception as a failed case, and `:160` sets `"passed": score >= contract.threshold`. A contract with a crashed case therefore passes when its other cases meet the threshold. The dict has no `inconclusive` key, unlike `TestResult` (`qualification.py:84-92`, BF-711) and the communication benchmarks (`communication_benchmarks.py:306-309`: "an unmeasured run does not pass and does not fail").
- #1123 asks for a statistical policy settled before any score blocks main, and for skipped, absent-data, judge-unavailable, contaminated or incomparable runs to stay non-passing. No catalog or policy exists (§11, Absence Verified).
- The Captain's trial rules (#1123 comment, 2026-10-20) bind every Ship Trial with crew. No schema carries them.

## 2. Solution

- `evaluate_contract`'s result gains `inconclusive` (true when any invoker call raised). `passed` is false whenever `inconclusive` is true. The dict's keys then equal `TestResult`'s fields, so the existing result model holds it and `QualificationStore` persists it.
- `_cmd_qa_run_contracts`'s stub raises `RuntimeError`, so no case is measured. Exit codes: 2 for a missing path (unchanged); otherwise 1 if a file fails to load or a measured contract fails; otherwise 3 if a contract was not measured or no file was found; otherwise 0. The table's last column is `Result` (`yes`, `no`, `not measured`), and an unmeasured contract shows `-` for its score.
- `tests/trials/catalog.py` loads and validates `tests/trials/data/catalog_v1.yaml`. It hashes the bytes it parses (AD-1143's convention), refuses unknown or missing keys in the document, the policy and each entry, refuses any policy mode except `report_only`, and requires the issue's non-passing statuses and fingerprint fields. It requires the Captain's rules verbatim in `crew_rules`. A task whose subject is `crew` must carry a disclosure (`what`, `ends`), `episodes: trial_labelled`, and `trust: separate` or `not_applied`. The catalog ships with no task. Production code imports none of it.

- **D1 Slice order:** S1 honest instruments and policy; S2 the disclosure and provenance seam; S3 AD-713-1 isolated invoker; S4 runner and report; S5 tasks and budgeted live runs (Captain: budget); S6 blocking policy (Captain).

### 7.4.N New file: `tests/test_ad1186_contract_inconclusive.py` (M1)

```python
"""AD-1186 (#1123) slice 1: an AD-713 contract run that measured nothing is inconclusive (BF-711).

A case whose invoker raised measured nothing, so the contract neither passes nor fails, and the
result carries AD-566a's ``TestResult`` fields exactly. ``probos qa run-contracts`` exits 3 for not
measured; 0 and 1 stay a measured pass and a measured failure (or a file that does not load).
"""

from __future__ import annotations

import argparse
import dataclasses
from pathlib import Path
from typing import Any

import pytest

from probos.cognitive import behavior_contract, qualification
from probos.cognitive.behavior_contract import (
    BehaviorContract,
    ContractCase,
    _MustRule,
    evaluate_contract,
)

_MUST_CONTRACT_YAML = (
    "name: needs_must\n"
    "agent: a\n"
    "threshold: 0.5\n"
    "cases:\n"
    "  - prompt: p\n"
    "    must:\n"
    "      - substring: ok\n"
)


def _contract() -> BehaviorContract:
    return BehaviorContract(
        name="t",
        agent="a",
        threshold=0.5,
        cases=[
            ContractCase(prompt="p1", must=[_MustRule(substring="ok")]),
            ContractCase(prompt="p2", must=[_MustRule(substring="ok")]),
        ],
    )


def _run_cli(path: Path) -> int:
    from probos.__main__ import _cmd_qa_run_contracts

    return _cmd_qa_run_contracts(argparse.Namespace(path=str(path)))


async def test_an_invoker_exception_makes_the_contract_inconclusive_and_not_passed() -> None:
    calls = 0

    async def invoker(agent_id: str, prompt: str) -> str:
        nonlocal calls
        calls += 1
        if calls == 1:
            raise RuntimeError("proxy down")
        return "ok"

    result = await evaluate_contract(_contract(), invoker)
    # The measured case passed, so the score (0.5) meets the threshold: before AD-1186 this read as a pass.
    assert result["score"] == 0.5
    assert result["inconclusive"] is True
    assert result["passed"] is False
    assert result["error"] == "proxy down"


async def test_a_measured_pass_and_a_measured_failure_are_not_inconclusive() -> None:
    async def agrees(agent_id: str, prompt: str) -> str:
        return "ok"

    async def disagrees(agent_id: str, prompt: str) -> str:
        return "no"

    passed = await evaluate_contract(_contract(), agrees)
    failed = await evaluate_contract(_contract(), disagrees)
    assert (passed["passed"], passed["inconclusive"]) == (True, False)
    assert (failed["passed"], failed["inconclusive"]) == (False, False)


async def test_the_result_is_a_qualification_test_result_and_round_trips_the_store(tmp_path: Path) -> None:
    async def unreachable(agent_id: str, prompt: str) -> str:
        raise RuntimeError("proxy down")

    result = await evaluate_contract(_contract(), unreachable)
    assert set(result) == {item.name for item in dataclasses.fields(qualification.TestResult)}
    store = qualification.QualificationStore(data_dir=tmp_path)
    await store.start()
    try:
        await store.save_result(qualification.TestResult(**result))
        stored = await store.get_latest("a", "t")
    finally:
        await store.stop()
    assert stored is not None
    assert stored.inconclusive is True
    assert stored.passed is False
    assert stored.error == "proxy down"


def test_the_cli_exits_one_when_a_file_fails_to_load_beside_an_unmeasured_contract(tmp_path: Path) -> None:
    (tmp_path / "a_measurable.yaml").write_text(_MUST_CONTRACT_YAML, encoding="utf-8")
    (tmp_path / "b_broken.yaml").write_text("name: no_agent_no_cases\n", encoding="utf-8")
    # A file that does not load is a failure; the unmeasured contract beside it does not mask it.
    assert _run_cli(tmp_path) == 1


@pytest.mark.parametrize(
    ("passed", "expected"),
    [(True, 0), (False, 1)],
    ids=["measured-pass", "measured-fail"],
)
def test_the_cli_maps_measured_verdicts_to_zero_and_one(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch, passed: bool, expected: int
) -> None:
    """The stub cannot measure anything, so the evaluator is replaced here: this pins the exit
    codes of a measured pass and a measured failure."""
    (tmp_path / "c.yaml").write_text(_MUST_CONTRACT_YAML, encoding="utf-8")

    async def measured(contract: Any, invoke_agent: Any) -> dict[str, Any]:
        return {"score": 1.0 if passed else 0.0, "passed": passed, "inconclusive": False}

    monkeypatch.setattr(behavior_contract, "evaluate_contract", measured)
    assert _run_cli(tmp_path) == expected
```

```python
def _mapping(value: Any, allowed: frozenset[str], required: frozenset[str], where: str) -> dict[str, Any]:
    """Return ``value`` as a mapping with no unknown and no missing key."""
    if not isinstance(value, dict):
        raise ValueError(f"{where}: expected a mapping")
    # YAML admits non-string keys; ordering by ``repr`` lists keys of mixed types.
    unknown = sorted(set(value) - allowed, key=repr)
    if unknown:
        raise ValueError(f"{where}: unknown keys {unknown}")
    absent = sorted(required - set(value))
    if absent:
        raise ValueError(f"{where}: missing keys {absent}")
    return value


def _names(value: Any, where: str) -> frozenset[str]:
    """Return ``value`` as a set of names; it must be a list of strings."""
    if not isinstance(value, list) or not all(isinstance(item, str) for item in value):
        raise ValueError(f"{where} must be a list of names")
    return frozenset(value)


def _one_of(value: Any, allowed: frozenset[str]) -> bool:
    """Return whether ``value`` is a string in ``allowed``.

    A YAML list or mapping gives ``False`` here, so the caller raises its ``ValueError``; testing
    membership directly would hash it and raise ``TypeError``.
    """
    return isinstance(value, str) and value in allowed


def validate_policy(value: Any, *, source: str) -> TrialPolicy:
    """Raise ``ValueError`` on any breach of the v1 policy."""
    policy = _mapping(value, _POLICY_KEYS, _POLICY_KEYS, f"{source}: policy")
    for key, expected in POLICY_CONSTANTS.items():
        if policy[key] != expected:
            raise ValueError(f"{source}: policy.{key} must be {expected!r}, not {policy[key]!r}")
    iterations, alpha, seed = policy["iterations"], policy["alpha"], policy["seed"]
    if type(iterations) is not int or iterations < 1000:
        raise ValueError(f"{source}: policy.iterations must be an integer of at least 1000")
    if type(alpha) is not float or not 0.0 < alpha < 1.0:
        raise ValueError(f"{source}: policy.alpha must be a float between 0 and 1")
    if type(seed) is not int:
        raise ValueError(f"{source}: policy.seed must be an integer")
    non_passing = _names(policy["non_passing"], f"{source}: policy.non_passing")
    dropped = sorted(REQUIRED_NON_PASSING - non_passing)
    if dropped:
        raise ValueError(f"{source}: policy.non_passing lacks {dropped}")
    fingerprint = _names(policy["fingerprint"], f"{source}: policy.fingerprint")
    unpinned = sorted(REQUIRED_FINGERPRINT - fingerprint)
    if unpinned:
        raise ValueError(f"{source}: policy.fingerprint lacks {unpinned}")
    return TrialPolicy(
        mode=policy["mode"],
        iterations=iterations,
        alpha=alpha,
        seed=seed,
        non_passing=non_passing,
        fingerprint=fingerprint,
    )


def validate_entry(value: Any, *, source: str) -> TrialEntry:
    """Raise ``ValueError`` on any breach; a crew task must carry rule 1's and rule 3's terms.

    A shape breach (not a mapping, an unknown or a missing key) is reported without the task's
    id, and an id that is not wholly a lower-case slug is reported by its ``repr``; every later
    message begins ``<source>: entry <id>:``.
    """
    entry = _mapping(value, _ENTRY_KEYS, _ENTRY_REQUIRED, f"{source}: entry")
    entry_id = entry["id"]
    if not isinstance(entry_id, str) or not _ENTRY_ID.fullmatch(entry_id):
        raise ValueError(f"{source}: entry id {entry_id!r} is not a lower-case slug")
    where = f"{source}: entry {entry_id}"
    if not _one_of(entry["category"], CATEGORIES):
        raise ValueError(f"{where}: unknown category {entry['category']!r}")
    if not _one_of(entry["instrument"], INSTRUMENTS):
        raise ValueError(f"{where}: unknown instrument {entry['instrument']!r}")
    ref = entry["ref"]
    if not isinstance(ref, str) or not ref.strip():
        raise ValueError(f"{where}: ref must name the instrument's contract, test or goal set")
    if not _one_of(entry["subject"], SUBJECTS):
        raise ValueError(f"{where}: unknown subject {entry['subject']!r}")
    common = {
        "id": entry_id,
        "category": entry["category"],
        "instrument": entry["instrument"],
        "ref": ref,
        "subject": entry["subject"],
    }
    if entry["subject"] != "crew":
        return TrialEntry(**common)
    disclosure = entry.get("disclosure")
    if not isinstance(disclosure, dict) or set(disclosure) != {"what", "ends"}:
        raise ValueError(f"{where}: a crew task needs a disclosure of exactly 'what' and 'ends' (rule 1)")
    what, ends = disclosure["what"], disclosure["ends"]
    if not all(isinstance(text, str) and text.strip() for text in (what, ends)):
        raise ValueError(f"{where}: the disclosure must say what the trial is and when it ends (rule 1)")
    if entry.get("episodes") != TRIAL_LABELLED:
        raise ValueError(f"{where}: a crew task's episodes must be {TRIAL_LABELLED!r} (rule 3)")
    if not _one_of(entry.get("trust"), TRUST_POLICIES):
        raise ValueError(f"{where}: a crew task's trust must be one of {sorted(TRUST_POLICIES)} (rule 3)")
    return TrialEntry(
        **common,
        disclosure_what=what,
        disclosure_ends=ends,
        episodes=entry["episodes"],
        trust=entry["trust"],
    )


def load_catalog(path: Path = CATALOG_V1_PATH) -> TrialCatalog:
    """Load, validate and hash one catalog file; the hash covers the bytes parsed."""
    raw = path.read_bytes()
    source = path.name
    document = _mapping(yaml.safe_load(raw), _TOP_KEYS, _TOP_KEYS, source)
    version = document["catalog_version"]
    if type(version) is not int or version != SUPPORTED_VERSION:
        raise ValueError(f"{source}: catalog_version must be {SUPPORTED_VERSION}")
    if document["crew_rules"] != CREW_RULES:
        raise ValueError(f"{source}: crew_rules must state the Captain's trial rules exactly (#1123)")
    policy = validate_policy(document["policy"], source=source)
    if not isinstance(document["entries"], list):
        raise ValueError(f"{source}: entries must be a list")
    entries = tuple(validate_entry(item, source=source) for item in document["entries"])
    ids = [entry.id for entry in entries]
    duplicates = sorted({entry_id for entry_id in ids if ids.count(entry_id) > 1})
    if duplicates:
        raise ValueError(f"{source}: duplicate entry ids {duplicates}")
    return TrialCatalog(
        version=version,
        sha256=hashlib.sha256(raw).hexdigest(),
        path=path,
        policy=policy,
        entries=entries,
    )
```
