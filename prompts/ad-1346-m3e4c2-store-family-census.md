class _CallScan(ast.NodeVisitor):
    def __init__(self, write: Callable[[ast.Call], str]) -> None:
        self.write = write
        self.scope: list[str] = []
        self.points: list[tuple[str, int, str]] = []

    def _enter(self, node: ast.ClassDef | ast.FunctionDef | ast.AsyncFunctionDef) -> None:
        self.scope.append(node.name)
        self.generic_visit(node)
        self.scope.pop()

    visit_ClassDef = visit_FunctionDef = visit_AsyncFunctionDef = _enter

    def visit_Call(self, node: ast.Call) -> None:
        name = self.write(node)
        if name:
            self.points.append((".".join(self.scope) or "<module>", node.lineno, name))
        self.generic_visit(node)

def _functions(path: str, qualified: str) -> list[ast.FunctionDef | ast.AsyncFunctionDef]:
    """Each function in one ``src`` module whose enclosing classes and functions and own name, joined by dots, are
    ``qualified``, as the scans join a scope."""
    found: list[ast.FunctionDef | ast.AsyncFunctionDef] = []
    stack: list[tuple[ast.AST, str]] = [(ast.parse(src_sources()[path], filename=path), "")]
    while stack:
        node, prefix = stack.pop()
        for child in ast.iter_child_nodes(node):
            if isinstance(child, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
                name = f"{prefix}.{child.name}" if prefix else child.name
                if name == qualified and not isinstance(child, ast.ClassDef):
                    found.append(child)
                stack.append((child, name))
            else:
                stack.append((child, prefix))
    return found
